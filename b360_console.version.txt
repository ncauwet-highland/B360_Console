# Windows version resource for b360_console.exe, consumed by PyInstaller's
# --version-file.
#
# This is not cosmetic. An executable with blank Properties -> Details is exactly
# what an unidentified binary looks like, which is the first thing a cautious
# customer or an AV heuristic notices. Filling it in costs nothing and is the
# other half of signing: the signature proves who built it, this says what it is.
#
# To roll the version, change BOTH filevers/prodvers and the two string values.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,          # VOS_NT_WINDOWS32
    fileType=0x1,        # VFT_APP
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',      # US English, Unicode
        [StringStruct('CompanyName', 'Highland Technology'),
         StringStruct('FileDescription', 'B360 Console - terminal and waveform viewer'),
         StringStruct('FileVersion', '1.0.0.0'),
         StringStruct('InternalName', 'b360_console'),
         StringStruct('LegalCopyright', 'Copyright (c) Highland Technology, Inc.'),
         StringStruct('OriginalFilename', 'b360_console.exe'),
         StringStruct('ProductName', 'B360 Console'),
         StringStruct('ProductVersion', '1.0.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [0x0409, 1200])])
  ]
)
